```typescript
  setSubmitError('Failed to submit claim. Please check your connection and try again.');
  // Keep CAPTCHA token so user doesn't have to reverify

```

#### 4.2: Timeout Handling
Add timeout for API calls:
```typescript
const controller = new AbortController();
const timeout = setTimeout(() => controller.abort(), 30000); // 30 second timeout

try {
  const response = await fetch('/api/public/claims/submit', {
    // ... other options
    signal: controller.signal,
  });
} catch (error) {
  if (error instanceof TypeError && error.message === 'Failed to fetch') {
    setSubmitError('Network timeout. Please check your connection and try again.');
  }
} finally {
  clearTimeout(timeout);
}
```
